ALGORITHM: ML-Based KV Cache Size Predictor for LLM Serving

=============================================================
PHASE 1: DATA COLLECTION
=============================================================

FUNCTION collect_training_data(vllm_engine, dataset):
    training_data = []
    
    FOR each request in dataset:
        
        # Extract input features BEFORE generation
        features = {
            input_length     ← count tokens in prompt,
            max_tokens       ← user specified parameter,
            temperature      ← user specified parameter,
            task_type        ← classify(prompt),
            hour_of_day      ← current hour
        }
        
        # Run normal vLLM generation
        output = vllm_engine.generate(request)
        
        # Record ground truth AFTER generation
        actual_blocks_used = output.kv_blocks_allocated
        actual_output_length = len(output.tokens)
        
        training_data.append(features, actual_blocks_used)
    
    RETURN training_data

FUNCTION classify_task(prompt_text):
    IF len(prompt_text) < 50:
        RETURN "short_query"
    ELSE IF "translate" in prompt_text:
        RETURN "translation"
    ELSE IF "```" in prompt_text:
        RETURN "code"
    ELSE:
        RETURN "general_chat"

=============================================================
PHASE 3: PREDICTION AT SERVING TIME
=============================================================

FUNCTION predict_kv_blocks(request, model, safety_level=0.85):
    
    features = extract_features(request)
    predicted_blocks = model[safety_level].predict(features)
    
    RETURN ceil(predicted_blocks)

=============================================================
PHASE 4: PREDICTIVE SCHEDULER
=============================================================

FUNCTION schedule_batch(pending_requests, available_blocks, model):
    
    # Step 1: Predict memory for each waiting request
    FOR each request in pending_requests:
        request.predicted_blocks = predict_kv_blocks(request, model)
    
    # Step 2: Sort by predicted size (smallest first)
    pending_requests.sort_by(predicted_blocks, ascending=True)
    
    # Step 3: Greedily pack requests into batch
    batch = []
    memory_used = 0
    
    FOR each request in pending_requests:
        IF memory_used + request.predicted_blocks <= available_blocks:
            batch.append(request)
            memory_used += request.predicted_blocks
        ELSE:
            BREAK   # remaining requests are larger, skip them
    
    RETURN batch

=============================================================
PHASE 5: PREDICTIVE BLOCK ALLOCATOR
=============================================================

FUNCTION allocate_with_prediction(request, model, block_allocator):
    
    # Pre-allocate predicted number of blocks upfront
    predicted_blocks = predict_kv_blocks(request, model)
    block_allocator.pre_warm(request.id, predicted_blocks)
    
    # Generation runs normally
    WHILE request not finished:
        
        token = generate_next_token(request)
        
        IF current_block is full:
            IF pre_allocated_blocks_available:
                # Fast path — use pre-allocated block
                use_next_preallocated_block()
            ELSE:
                # Prediction was too low — fallback gracefully
                block_allocator.allocate_new_block()
        
    # If we over-predicted, free unused blocks
    block_allocator.free_unused(request.id)

=============================================================
PHASE 6: ONLINE LEARNING (OPTIONAL IMPROVEMENT)
=============================================================

FUNCTION update_model_online(request, actual_blocks_used):
    
    # After each request completes, log the result
    new_datapoint = (request.features, actual_blocks_used)
    training_buffer.append(new_datapoint)
    
    # Periodically retrain model on accumulated data
    IF len(training_buffer) >= RETRAIN_THRESHOLD:
        retrain_predictor(training_buffer)
        training_buffer.clear()

=============================================================
MAIN SERVING LOOP
=============================================================

FUNCTION main():
    
    # Startup
    vllm_engine = initialize_vllm()
    training_data = collect_training_data(vllm_engine, ShareGPT_dataset)
    model = train_predictor(training_data)
    
    # Serving
    WHILE serving:
        
        pending_requests = get_waiting_requests()
        available_blocks = gpu_block_allocator.free_blocks()
        
        # Use ML model to schedule intelligently
        batch = schedule_batch(pending_requests, available_blocks, model)
        
        FOR each request in batch:
            allocate_with_prediction(request, model, block_allocator)
            execute(request)
            update_model_online(request, actual_blocks_used)
